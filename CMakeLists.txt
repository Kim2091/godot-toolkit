cmake_minimum_required(VERSION 3.24)
project(GodotRemixEditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(REMIX_EDITOR_GODOT "Build the Godot extension" ON)
option(REMIX_EDITOR_TESTS "Build native workflow tests" ON)
set(GODOT_CPP_DIR "${CMAKE_SOURCE_DIR}/.work/deps/godot-cpp" CACHE PATH "godot-cpp checkout")
set(REMIX_SOURCE_DIR "${CMAKE_SOURCE_DIR}/ref/dxvk-remix-gmod" CACHE PATH "Pinned Remix checkout")
find_package(pxr CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
find_package(assimp CONFIG REQUIRED)
add_library(editor_document native/document.cpp native/authoring.cpp native/scene.cpp native/ingestion.cpp native/channel.cpp)
target_include_directories(editor_document PUBLIC native ${PXR_INCLUDE_DIRS})
target_link_libraries(editor_document PUBLIC usdGeom usdShade usdLux usdUtils nlohmann_json::nlohmann_json Threads::Threads)
if(assimp_FOUND)
  target_link_libraries(editor_document PRIVATE assimp::assimp)
  target_compile_definitions(editor_document PRIVATE EDITOR_WITH_ASSIMP=1)
endif()
add_executable(remix-editor-cli native/cli.cpp)
target_link_libraries(remix-editor-cli PRIVATE editor_document)
add_executable(remix-renderer native/renderer_main.cpp native/channel.cpp)
target_include_directories(remix-renderer PRIVATE native "${REMIX_SOURCE_DIR}/public/include")
target_link_libraries(remix-renderer PRIVATE nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(remix-renderer PRIVATE user32)
endif()
if(REMIX_EDITOR_GODOT)
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "USD document errors are caught at the extension boundary" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_SOURCE_DIR}/native/godot_build_profile.json" CACHE FILEPATH "Godot binding subset")
  add_subdirectory("${GODOT_CPP_DIR}" godot-cpp EXCLUDE_FROM_ALL)
  find_package(Vulkan REQUIRED)
  add_library(remix_editor SHARED native/godot_extension.cpp native/gpu_bridge.cpp)
  target_link_libraries(remix_editor PRIVATE editor_document godot::cpp Vulkan::Vulkan)
  set_target_properties(remix_editor PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app/bin"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app/bin"
    BUILD_RPATH "${PXR_LIB_DIR}")
endif()
if(REMIX_EDITOR_TESTS)
  enable_testing()
  add_executable(render_settings_tests tests/render_settings_tests.cpp)
  target_include_directories(render_settings_tests PRIVATE native)
  target_link_libraries(render_settings_tests PRIVATE nlohmann_json::nlohmann_json)
  add_test(NAME render_settings COMMAND render_settings_tests)
  add_executable(document_tests tests/document_tests.cpp)
  target_link_libraries(document_tests PRIVATE editor_document)
  target_compile_definitions(document_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
  add_test(NAME document_workflows COMMAND document_tests)
  add_executable(authoring_tests tests/authoring_tests.cpp)
  target_link_libraries(authoring_tests PRIVATE editor_document)
  target_compile_definitions(authoring_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
  add_test(NAME authoring_workflows COMMAND authoring_tests)
  add_executable(placement_tests tests/placement_tests.cpp)
  target_link_libraries(placement_tests PRIVATE editor_document)
  target_compile_definitions(placement_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
  add_test(NAME placement_workflows COMMAND placement_tests)
  add_executable(light_updates_tests tests/light_updates_tests.cpp)
  target_link_libraries(light_updates_tests PRIVATE editor_document)
  target_compile_definitions(light_updates_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
  add_test(NAME light_updates COMMAND light_updates_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME renderer_protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/protocol_tests.py" $<TARGET_FILE:remix-renderer>)
endif()
install(TARGETS remix-editor-cli remix-renderer RUNTIME DESTINATION bin)
